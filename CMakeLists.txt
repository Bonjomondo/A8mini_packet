cmake_minimum_required(VERSION 3.16)
project(A8MiniProbe VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt5 5.15 REQUIRED COMPONENTS Core Gui Qml Quick QuickControls2 Network)
find_package(PkgConfig REQUIRED)
pkg_check_modules(GST REQUIRED IMPORTED_TARGET gstreamer-1.0>=1.18 gstreamer-video-1.0)
set(A8_QMLGL_SOURCE_DIR "" CACHE PATH "Optional matching gst-plugins-good source tree to build a Qt5 qmlgl plugin")
if(A8_QMLGL_SOURCE_DIR)
    add_subdirectory(cmake/qmlgl)
endif()
add_library(a8video STATIC src/a8mini/A8VideoPipeline.cpp src/a8mini/A8VideoPipeline.h)
target_include_directories(a8video PUBLIC src/a8mini)
target_link_libraries(a8video PUBLIC Qt5::Quick PkgConfig::GST)
add_library(a8gimbal STATIC src/a8mini/A8GimbalProtocol.cpp src/a8mini/A8GimbalProtocol.h
    src/a8mini/A8GimbalController.cpp src/a8mini/A8GimbalController.h)
target_include_directories(a8gimbal PUBLIC src/a8mini)
target_link_libraries(a8gimbal PUBLIC Qt5::Core Qt5::Network)
option(A8_BUILD_CONTROL_TESTS "Build gimbal protocol and transport tests" OFF)
if(A8_BUILD_CONTROL_TESTS)
    enable_testing()
    find_package(Qt5 REQUIRED COMPONENTS Test)
    add_executable(a8gimbal_tests tests/a8mini/test-gimbal.cpp)
    target_link_libraries(a8gimbal_tests PRIVATE a8gimbal Qt5::Test)
    add_test(NAME a8gimbal_tests COMMAND a8gimbal_tests)
endif()
add_subdirectory(tools/a8mini_probe)
